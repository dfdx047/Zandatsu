# M0: the lifted MGR:R PC exe + the shared runtime as a Linux x86-64 executable, D3D9 through the same DXVK 1.9.2b
# build Android uses (host flavour, Xlib test window). Adapted from codepdbh/nfsu2-recompiled tests/android_runtime (MIT).
cmake_minimum_required(VERSION 3.20)
project(MGRPCLinux C CXX)
get_filename_component(PORT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
get_filename_component(ROOT "${PORT}/../.." ABSOLUTE)
set(LAB "${ROOT}/build/lab/pc")
set(RT "${LAB}/pcrecomp-pinned/runtime")
set(GEN "${ROOT}/build/pc/gen" CACHE PATH "lift output (run_lift.py lift)")
set(DXVK_SOURCE "${LAB}/dxvk-source" CACHE PATH "prepared DXVK")
set(DXVK_LIBRARY "${LAB}/dxvk-host/libdxvk_android.so" CACHE FILEPATH "host DXVK")
set(GUEST_OPT "-O0" CACHE STRING "optimisation of the generated C")
file(GLOB GEN_SOURCES "${GEN}/recomp_*.c")
add_library(guest OBJECT ${GEN_SOURCES})
target_include_directories(guest PRIVATE "${RT}/recomp32" "${GEN}" "${PORT}")
target_compile_definitions(guest PRIVATE RECOMP_STRICT_ICALL)
target_compile_options(guest PRIVATE ${GUEST_OPT} -w -fno-strict-aliasing -fexceptions -ffp-contract=off -include "${PORT}/arm64_guest.h")
add_executable(mgr_pc main.cpp "${PORT}/guest_runtime.cpp" "${PORT}/overrides.c" "${PORT}/d3d9_bridge.cpp" $<TARGET_OBJECTS:guest>)
target_compile_features(mgr_pc PRIVATE cxx_std_17)
target_compile_definitions(mgr_pc PRIVATE RECOMP_STRICT_ICALL NFS_D3D9_BACKEND)
target_include_directories(mgr_pc PRIVATE "${PORT}" "${RT}/recomp32" "${RT}/win32hle" "${GEN}")
target_include_directories(mgr_pc SYSTEM PRIVATE "${DXVK_SOURCE}/include" "${DXVK_SOURCE}/include/native/windows" "${DXVK_SOURCE}/include/native/directx")
add_library(native_d3d9 SHARED IMPORTED)
set_target_properties(native_d3d9 PROPERTIES IMPORTED_LOCATION "${DXVK_LIBRARY}")
target_link_libraries(mgr_pc PRIVATE native_d3d9 m pthread)

# Local, untracked runtime additions (never in git), e.g. -DMGR_LAB_INCLUDE=/abs/path/lab.inc
set(MGR_LAB_INCLUDE "" CACHE FILEPATH "untracked lab include for guest_mgr.inc")
if(MGR_LAB_INCLUDE)
  target_compile_definitions(mgr_pc PRIVATE "MGR_LAB_INCLUDE=\"${MGR_LAB_INCLUDE}\"")
endif()
